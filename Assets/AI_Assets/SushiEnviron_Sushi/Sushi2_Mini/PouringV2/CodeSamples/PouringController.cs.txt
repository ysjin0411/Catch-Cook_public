using System;
using UnityEngine;
using UnityEngine.InputSystem;

namespace SushiPouringSample
{
    /// <summary>입력, 모델 갱신, 표시 갱신과 결과 전달을 연결합니다.</summary>
    public sealed class PouringController : MonoBehaviour
    {
        [SerializeField] private PourSettings _settings = new PourSettings();
        [SerializeField] private PouringView _view;
        [SerializeField] private InputActionReference _tapAction;
        [SerializeField] private InputActionReference _cancelAction;
        private InputAction _ownedTap;
        private InputAction _ownedCancel;
        private PouringModel _model;
        private bool _armed;
        private bool _reported;
        private int _startFrame;
        public bool IsRunning => _model != null && _model.Result == PourResult.Running;
        public event Action<PourResult> Completed;

        /// <summary>상호작용 대상이 준비된 뒤 새 음료 미니게임을 시작합니다.</summary>
        public void Begin(Color drinkColor)
        {
            // 중복 시작과 비활성 상태에서의 시작을 차단합니다.
            if (IsRunning || !isActiveAndEnabled) return;
            // 필요한 UI와 입력을 연결하지 않았다면 명확한 오류를 남깁니다.
            if (!_view || !_view.IsConfigured() || !_tapAction || _tapAction.action == null)
            {
                // 실패한 시작에는 보상이나 완료 이벤트를 발생시키지 않습니다.
                Debug.LogError("PouringView와 Tap Action 연결을 확인하세요.", this);
                return;
            }
            // 설정 검증을 통과한 세션을 생성합니다.
            _model = new PouringModel(_settings);
            _reported = false;
            _armed = false;
            _startFrame = Time.frameCount;
            // 공유 액션을 끄지 않도록 미니게임이 소유하는 복제본을 활성화합니다.
            ReleaseInput();
            _ownedTap = _tapAction.action.Clone();
            _ownedTap.Enable();
            if (_cancelAction && _cancelAction.action != null)
            {
                // 취소 입력도 공유 ActionMap과 수명을 분리합니다.
                _ownedCancel = _cancelAction.action.Clone();
                _ownedCancel.Enable();
            }
            // 라운드 시작 시 이전 수위와 효과를 초기 표시합니다.
            _view.Configure(_settings, drinkColor, _model);
            _view.Render(_model);
        }

        /// <summary>새로 누른 입력을 읽고 시간에 따라 모델과 UI를 갱신합니다.</summary>
        private void Update()
        {
            // 진행 중인 세션이 없거나 시작한 프레임이면 입력을 처리하지 않습니다.
            if (!IsRunning || Time.frameCount == _startFrame) return;
            // 게임 일시정지 중에는 입력과 시뮬레이션을 함께 멈춥니다.
            if (Time.timeScale <= 0f) return;
            // 취소 입력은 따르기 입력보다 먼저 처리합니다.
            if (_ownedCancel != null && _ownedCancel.WasPressedThisFrame())
            {
                // 취소도 같은 결과 전달 경로를 사용합니다.
                Cancel();
                return;
            }
            // UI를 연 키를 한 번 놓아야 연타를 시작할 수 있습니다.
            if (!_armed)
            {
                // 시작 입력이 액션 복제본에 중복 반영되는 것을 막습니다.
                _armed = !_ownedTap.IsPressed();
            }
            else if (_ownedTap.WasPressedThisFrame())
            {
                // 누르고 있기 대신 새로운 누름만 한 번의 충격으로 전달합니다.
                _model.Tap();
            }
            // 모델 내부가 시간을 작게 분할하므로 렌더 프레임과 계산 단위를 분리합니다.
            _model.Tick(Time.deltaTime);
            _view.Render(_model);
            ReportIfFinished();
        }

        /// <summary>손님 이탈, 주문 변경, 거리 이탈 또는 닫기 요청을 처리합니다.</summary>
        public void Cancel()
        {
            // 이미 종료된 세션에는 아무 변화도 주지 않습니다.
            if (!IsRunning) return;
            // 모델과 표시를 먼저 종료한 뒤 구독자에게 결과를 보냅니다.
            _model.Cancel();
            if (_view && _view.IsConfigured()) _view.Render(_model);
            ReportIfFinished();
        }

        /// <summary>결과를 정확히 한 번 전달하고 소유 입력을 해제합니다.</summary>
        private void ReportIfFinished()
        {
            // 중복 보상이나 중복 종료 콜백을 방지합니다.
            if (_model == null || _model.Result == PourResult.Running || _reported) return;
            _reported = true;
            ReleaseInput();
            // 콜백에서 패널을 닫더라도 이미 완료 보고 여부가 설정되어 있습니다.
            Completed?.Invoke(_model.Result);
        }

        /// <summary>미니게임이 복제하여 소유한 입력만 폐기합니다.</summary>
        private void ReleaseInput()
        {
            // 공유 플레이어 입력의 Enable 상태는 변경하지 않습니다.
            _ownedTap?.Dispose();
            _ownedCancel?.Dispose();
            _ownedTap = null;
            _ownedCancel = null;
        }

        /// <summary>패널 비활성화 시 미완료 세션과 입력을 정리합니다.</summary>
        private void OnDisable()
        {
            // 종료 콜백에는 Cancelled가 한 번만 전달됩니다.
            Cancel();
            // FluidRig가 먼저 취소했더라도 종료 결과를 빠뜨리지 않습니다.
            ReportIfFinished();
            ReleaseInput();
        }

        /// <summary>앱 포커스 상실로 발생할 수 있는 입력 누락을 취소로 처리합니다.</summary>
        private void OnApplicationFocus(bool focused)
        {
            // 창 밖에서 시간이 흘러 의도치 않게 넘치는 상황을 방지합니다.
            if (!focused) Cancel();
        }

        /// <summary>플레이 모드에서 연결 상태를 확인할 수 있는 테스트 라운드입니다.</summary>
        [ContextMenu("Test / Begin Lemon Pour")]
        private void BeginTestRound()
        {
            // 에디터 편집 중에는 입력을 생성하지 않습니다.
            if (Application.isPlaying) Begin(new Color(1f, 0.82f, 0.22f, 1f));
        }
    }
}
