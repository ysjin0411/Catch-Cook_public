using System;

namespace SushiPouringV3
{
    /// <summary>입자 하나의 위치, 속도와 보존되는 양을 저장합니다.</summary>
    public struct FluidParticle
    {
        public double X, Y, VX, VY, Volume;
        public bool Entered;
        internal double OldX, OldY;
    }

    /// <summary>Unity 물리 엔진과 분리된 소규모 정면 유체 입자 솔버입니다.</summary>
    public sealed class FluidSimulation
    {
        public const int Capacity = 384;
        public const double Bottom = -2.4, Rim = 0.4;
        public const double BottomWidth = 0.72, TopWidth = 0.84;
        public const double CupArea = (BottomWidth + TopWidth) * (Rim - Bottom);
        public const double ParticleArea = 0.021;
        public const double Parcel = ParticleArea / CupArea;
        private const double Radius = 0.055, H = 0.28;
        private readonly FluidParticle[] _particles = new FluidParticle[Capacity];
        private readonly double[] _density = new double[Capacity];
        private readonly double[] _nearDensity = new double[Capacity];
        public int Count { get; private set; }
        public double Emitted { get; private set; }
        public double Spilled { get; private set; }

        /// <summary>표시 계층에 입자 상태의 복사본을 반환합니다.</summary>
        public FluidParticle GetParticle(int index)
        {
            // 외부에서 내부 배열을 수정하지 못하도록 값 타입을 반환합니다.
            if (index < 0 || index >= Count) throw new ArgumentOutOfRangeException(nameof(index));
            return _particles[index];
        }

        /// <summary>해당 높이에서 컵 안쪽의 반너비를 반환합니다.</summary>
        public static double HalfWidth(double y)
        {
            // 아래가 좁고 위가 넓은 기존 컵 이미지와 같은 경계입니다.
            double t = Math.Max(0d, Math.Min(1d, (y - Bottom) / (Rim - Bottom)));
            return BottomWidth + (TopWidth - BottomWidth) * t;
        }

        /// <summary>정규화된 높이를 컵 단면적 기준의 양으로 변환합니다.</summary>
        public static double VolumeAtHeight(double height)
        {
            // 테이퍼 컵의 아래부터 목표 높이까지 단면적을 적분합니다.
            double t = Math.Max(0d, Math.Min(1d, height));
            return (2d * BottomWidth * t + (TopWidth - BottomWidth) * t * t) / (BottomWidth + TopWidth);
        }

        /// <summary>보존되는 양으로부터 물결에 영향받지 않는 기준 높이를 구합니다.</summary>
        public static double HeightAtVolume(double volume)
        {
            // 컵 단면적 식의 양의 근을 계산합니다.
            double v = Math.Max(0d, volume);
            double a = TopWidth - BottomWidth;
            return (-BottomWidth + Math.Sqrt(BottomWidth * BottomWidth + a * v * (BottomWidth + TopWidth))) / a;
        }

        /// <summary>빈 슬롯이 있을 때만 새 입자를 발사합니다.</summary>
        public bool TryEmit(double x, double y, double vx, double vy, double volume)
        {
            // 풀 고갈 시 호출자가 예산을 보존하도록 실패를 반환합니다.
            if (Count >= Capacity || !(volume > 0d) || double.IsInfinity(volume)) return false;
            if (double.IsNaN(x + y + vx + vy) || double.IsInfinity(x + y + vx + vy)) return false;
            // 생성 시 양을 한 번만 기록합니다.
            _particles[Count++] = new FluidParticle { X=x, Y=y, VX=vx, VY=vy, Volume=volume };
            Emitted += volume;
            return true;
        }

        /// <summary>컵 안에 들어온 양, 공중의 양과 평균 운동량을 계산합니다.</summary>
        public void Measure(out double contained, out double flying, out double meanSpeed)
        {
            // 출렁이는 최고점을 사용하지 않고 입자별 양을 합산합니다.
            contained = 0d;
            flying = 0d;
            meanSpeed = 0d;
            for (int i = 0; i < Count; i++)
            {
                // 입구를 통과한 입자만 담긴 양에 포함합니다.
                FluidParticle p = _particles[i];
                if (p.Entered) contained += p.Volume;
                else flying += p.Volume;
                meanSpeed += Math.Sqrt(p.VX * p.VX + p.VY * p.VY);
            }
            // 입자가 없을 때는 안정 상태입니다.
            if (Count > 0) meanSpeed /= Count;
        }

        /// <summary>최대 1/120초의 한 스텝에서 점성, 밀도 압력과 컵 충돌을 계산합니다.</summary>
        public void Step(double dt)
        {
            // 큰 프레임은 호출자가 고정 스텝으로 나눠 전달합니다.
            if (!(dt > 0d) || dt > 1d / 120d + 1e-9) throw new ArgumentOutOfRangeException(nameof(dt));
            // 이웃끼리 속도를 조금씩 공유하여 점성을 표현합니다.
            for (int i = 0; i < Count; i++)
            for (int j = i + 1; j < Count; j++)
            {
                // 작은 입자 수를 대상으로 모든 쌍을 검사하는 O(N²) 구현입니다.
                double dx = _particles[j].X - _particles[i].X;
                double dy = _particles[j].Y - _particles[i].Y;
                double d2 = dx*dx + dy*dy;
                if (d2 >= H*H) continue;
                double mix = 1.8d * dt * (1d - Math.Sqrt(d2) / H);
                double ix = (_particles[j].VX - _particles[i].VX) * mix;
                double iy = (_particles[j].VY - _particles[i].VY) * mix;
                _particles[i].VX += ix; _particles[j].VX -= ix;
                _particles[i].VY += iy; _particles[j].VY -= iy;
            }
            for (int i = 0; i < Count; i++)
            {
                // 이전 위치를 저장한 뒤 중력으로 예측 위치를 만듭니다.
                _particles[i].OldX = _particles[i].X;
                _particles[i].OldY = _particles[i].Y;
                _particles[i].VY -= 9.81d * dt;
                _particles[i].X += _particles[i].VX * dt;
                _particles[i].Y += _particles[i].VY * dt;
            }
            for (int iteration = 0; iteration < 2; iteration++)
            {
                // 보정마다 현재 이웃 밀도를 다시 계산합니다.
                Array.Clear(_density, 0, Count);
                Array.Clear(_nearDensity, 0, Count);
                for (int i = 0; i < Count; i++)
                for (int j = i + 1; j < Count; j++)
                {
                    // 가까울수록 큰 밀도와 근거리 밀도를 양쪽에 더합니다.
                    double dx = _particles[j].X - _particles[i].X;
                    double dy = _particles[j].Y - _particles[i].Y;
                    double d = Math.Sqrt(dx*dx + dy*dy);
                    if (d >= H) continue;
                    double q = 1d - d/H;
                    _density[i] += q*q; _density[j] += q*q;
                    _nearDensity[i] += q*q*q; _nearDensity[j] += q*q*q;
                }
                for (int i = 0; i < Count; i++)
                for (int j = i + 1; j < Count; j++)
                {
                    // 밀도 압력과 근거리 압력으로 두 입자의 위치를 반대 방향으로 보정합니다.
                    double dx = _particles[j].X - _particles[i].X;
                    double dy = _particles[j].Y - _particles[i].Y;
                    double d = Math.Sqrt(dx*dx + dy*dy);
                    if (d >= H) continue;
                    if (d < 1e-7) { dx = 1e-7; dy = 0d; d = 1e-7; }
                    double q = 1d - d/H;
                    double pressure = 140d * ((_density[i] + _density[j]) * 0.5d - 1.6d);
                    pressure = Math.Max(-8d, pressure);
                    double near = 400d * (_nearDensity[i] + _nearDensity[j]) * 0.5d;
                    double shift = dt*dt * (pressure*q + near*q*q) * 0.5d;
                    // 극단적인 겹침은 최소 거리 제약으로 풀고 폭발적인 한 번의 이동은 제한합니다.
                    shift = Math.Max(shift, (0.11d - d) * 0.45d);
                    shift = Math.Max(-0.003d, Math.Min(0.018d, shift));
                    double sx = dx/d*shift, sy = dy/d*shift;
                    _particles[i].X -= sx; _particles[j].X += sx;
                    _particles[i].Y -= sy; _particles[j].Y += sy;
                }
                // 위치 보정으로 벽을 뚫지 않도록 매 반복 끝에 컵 경계를 적용합니다.
                for (int i = 0; i < Count; i++) ResolveCup(i);
            }
            for (int i = Count - 1; i >= 0; i--)
            {
                // 보정된 위치에서 다음 스텝의 속도를 복구합니다.
                _particles[i].VX = (_particles[i].X - _particles[i].OldX) / dt;
                _particles[i].VY = (_particles[i].Y - _particles[i].OldY) / dt;
                double speed = Math.Sqrt(_particles[i].VX*_particles[i].VX + _particles[i].VY*_particles[i].VY);
                double scale = Math.Exp(-0.25d*dt) * Math.Min(1d, 12d / Math.Max(12d, speed));
                _particles[i].VX *= scale; _particles[i].VY *= scale;
                // 열린 윗면으로 밖에 나가거나 화면 아래로 떨어진 입자는 손실 처리합니다.
                bool escaped = _particles[i].Entered && _particles[i].Y > Rim + Radius && Math.Abs(_particles[i].X) > TopWidth;
                if (escaped || _particles[i].Y < -3.2d || Math.Abs(_particles[i].X) > 2.8d)
                {
                    // 마지막 입자를 빈 슬롯으로 옮겨 배열을 조밀하게 유지합니다.
                    Spilled += _particles[i].Volume;
                    _particles[i] = _particles[--Count];
                }
            }
        }

        /// <summary>입자가 컵의 열린 입구로 들어오면 바닥과 기울어진 벽에 가둡니다.</summary>
        private void ResolveCup(int index)
        {
            // 옆에서 날아온 방울을 벽 안으로 순간이동시키지 않습니다.
            if (!_particles[index].Entered && _particles[index].Y <= Rim &&
                _particles[index].OldY > Rim && Math.Abs(_particles[index].X) < TopWidth - Radius)
                _particles[index].Entered = true;
            if (!_particles[index].Entered) return;
            // 물방울 중심을 실제 반지름만큼 바닥 위에 유지합니다.
            _particles[index].Y = Math.Max(Bottom + Radius, _particles[index].Y);
            if (_particles[index].Y >= Rim) return;
            // 기울어진 벽의 법선 방향으로 밀어내어 수직벽처럼 처리하지 않습니다.
            double slope = (TopWidth-BottomWidth)/(Rim-Bottom);
            double side = _particles[index].X < 0d ? -1d : 1d;
            double excess = Math.Abs(_particles[index].X) - (HalfWidth(_particles[index].Y)-Radius);
            if (excess <= 0d) return;
            double correction = excess/(1d+slope*slope);
            _particles[index].X -= side*correction;
            _particles[index].Y += slope*correction;
        }
    }
}
