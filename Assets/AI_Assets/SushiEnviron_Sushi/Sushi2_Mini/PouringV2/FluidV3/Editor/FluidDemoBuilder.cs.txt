#if UNITY_EDITOR
using UnityEditor;
using UnityEngine;

namespace SushiPouringV3.Editor
{
    /// <summary>사용자가 메뉴를 실행했을 때만 독립 테스트 프리팹을 생성합니다.</summary>
    public static class FluidDemoBuilder
    {
        private const string BasePath = "Assets/99.StoreAssets/SushiEnviron_Sushi/Sushi2_Mini/PouringV2";

        /// <summary>현재 씬을 저장하거나 바꾸지 않고 미니게임 프리팹을 만듭니다.</summary>
        [MenuItem("Tools/Sushi Mini/Create Fluid V3 Demo Prefab")]
        public static void CreatePrefab()
        {
            // 필요한 스프라이트만 직접 사용자가 요청한 메뉴 실행 시 임포트합니다.
            string[] names = { "Bottle_Opaque_Wide", "Cup_Back", "Cup_Front", "Gauge_Track", "Gauge_Triangle", "Target_Band", "Panel_Background" };
            foreach (string name in names)
            {
                // 이 메뉴는 지정된 리소스의 임포트 설정을 Sprite로 바꿉니다.
                string path = BasePath + "/Sprites/" + name + ".png";
                TextureImporter importer = AssetImporter.GetAtPath(path) as TextureImporter;
                if (!importer) throw new System.InvalidOperationException("리소스 없음: " + path);
                importer.textureType = TextureImporterType.Sprite;
                importer.spriteImportMode = SpriteImportMode.Single;
                importer.alphaIsTransparency = true;
                importer.mipmapEnabled = false;
                importer.textureCompression = TextureImporterCompression.Uncompressed;
                importer.SaveAndReimport();
            }
            // Shader.Find 대신 에셋 참조를 저장하여 빌드에서 셰이더가 누락되지 않게 합니다.
            Shader shader = AssetDatabase.LoadAssetAtPath<Shader>(BasePath + "/FluidV3/Code/UI_FluidDensity.shader");
            if (!shader) throw new System.InvalidOperationException("UI_FluidDensity.shader.txt의 마지막 .txt를 제거하세요.");
            // 임시 미리보기 씬에만 계층을 만들어 작업 중인 씬과 분리합니다.
            var previewScene = UnityEditor.SceneManagement.EditorSceneManager.NewPreviewScene();
            GameObject root = new GameObject("FluidV3_Demo", typeof(RectTransform), typeof(Canvas), typeof(UnityEngine.UI.CanvasScaler));
            UnityEngine.SceneManagement.SceneManager.MoveGameObjectToScene(root, previewScene);
            try
            {
                // 임시 제작 오브젝트는 현재 씬에 저장되지 않습니다.
                root.hideFlags = HideFlags.None;
                root.GetComponent<Canvas>().renderMode = RenderMode.ScreenSpaceOverlay;
                root.GetComponent<Canvas>().sortingOrder = 100;
                var scaler = root.GetComponent<UnityEngine.UI.CanvasScaler>();
                scaler.uiScaleMode = UnityEngine.UI.CanvasScaler.ScaleMode.ScaleWithScreenSize;
                scaler.referenceResolution = new Vector2(1920, 1080);
                scaler.matchWidthOrHeight = 0.5f;
                // 비활성 패널을 조립하므로 편집 중 라운드가 시작되지 않습니다.
                RectTransform panel = Rect("PourPanel", root.transform, new Vector2(500,600), Vector2.zero);
                panel.gameObject.SetActive(false);
                Image("Background", panel, "Panel_Background", new Vector2(600,780), Vector2.zero);
                RectTransform bottle = Rect("BottlePivot", panel, Vector2.zero, new Vector2(45,125));
                bottle.localRotation = Quaternion.Euler(0,0,150);
                var bottleArt = Image("BottleArt", bottle, "Bottle_Opaque_Wide", new Vector2(160,213.333f), Vector2.zero);
                bottleArt.rectTransform.pivot = new Vector2(0.5f,0.855f);
                RectTransform mouth = Rect("Mouth", bottle, Vector2.zero, Vector2.zero);
                Image("CupBack", panel, "Cup_Back", new Vector2(240,320), new Vector2(0,-100));
                RectTransform liquid = Rect("FluidOutput", panel, new Vector2(500,600), Vector2.zero);
                var raw = liquid.gameObject.AddComponent<UnityEngine.UI.RawImage>();
                raw.raycastTarget = false;
                Image("CupFront", panel, "Cup_Front", new Vector2(240,320), new Vector2(0,-100));
                // 목표 영역은 컵 내부 높이 -240~40의 68~78%에 맞춥니다.
                var target = Image("TargetBand", panel, "Target_Band", new Vector2(168,28), new Vector2(0,-35.6f));
                target.color = new Color(1,1,1,0.22f);
                Image("TargetLower", panel, "Target_Band", new Vector2(180,2), new Vector2(0,-49.6f));
                Image("TargetUpper", panel, "Target_Band", new Vector2(180,2), new Vector2(0,-21.6f));
                var gauge = Image("PressureTrack", panel, "Gauge_Track", new Vector2(400,24), new Vector2(0,-285));
                Image("IdealBand", gauge.transform, "Target_Band", new Vector2(80,24), Vector2.zero);
                var marker = Image("Triangle", gauge.transform, "Gauge_Triangle", new Vector2(24,18), new Vector2(-200,-18));
                RectTransform statusRect = Rect("Status", panel, new Vector2(540,50), new Vector2(0,-345));
                var status = statusRect.gameObject.AddComponent<UnityEngine.UI.Text>();
                status.font = Resources.GetBuiltinResource<Font>("LegacyRuntime.ttf");
                status.fontSize = 20; status.alignment = TextAnchor.MiddleCenter; status.color = Color.white;
                status.raycastTarget = false;
                // Presenter, View와 모델의 연결은 프리팹 참조로 저장합니다.
                FluidDensityView view = panel.gameObject.AddComponent<FluidDensityView>();
                Assign(view,"_output",raw); Assign(view,"_shader",shader);
                FluidPourPresenter presenter = panel.gameObject.AddComponent<FluidPourPresenter>();
                Assign(presenter,"_fluidView",view); Assign(presenter,"_fluidRect",liquid);
                Assign(presenter,"_bottlePivot",bottle); Assign(presenter,"_mouth",mouth);
                Assign(presenter,"_marker",marker.rectTransform); Assign(presenter,"_status",status);
                panel.gameObject.SetActive(true);
                // 기존 프리팹을 덮어쓰지 않고 새 경로를 만들어 저장합니다.
                string path = AssetDatabase.GenerateUniqueAssetPath(BasePath + "/FluidV3/FluidV3_Demo.prefab");
                GameObject prefab = PrefabUtility.SaveAsPrefabAsset(root,path);
                Selection.activeObject = prefab;
                EditorGUIUtility.PingObject(prefab);
            }
            finally
            {
                // 실패한 경우에도 임시 계층을 현재 씬에 남기지 않습니다.
                UnityEditor.SceneManagement.EditorSceneManager.ClosePreviewScene(previewScene);
            }
        }

        /// <summary>중앙 앵커와 크기를 갖는 UI 사각형을 만듭니다.</summary>
        private static RectTransform Rect(string name, Transform parent, Vector2 size, Vector2 position)
        {
            // 부모를 연결한 뒤 로컬 UI 좌표를 명시합니다.
            var rect = new GameObject(name,typeof(RectTransform)).GetComponent<RectTransform>();
            rect.SetParent(parent,false);
            rect.anchorMin = rect.anchorMax = rect.pivot = new Vector2(0.5f,0.5f);
            rect.sizeDelta=size; rect.anchoredPosition=position;
            return rect;
        }

        /// <summary>지정된 스프라이트를 사용하는 Image를 만듭니다.</summary>
        private static UnityEngine.UI.Image Image(string name, Transform parent, string sprite, Vector2 size, Vector2 position)
        {
            // 그림의 여백을 유지하므로 코드의 병 입구와 컵 좌표가 일치합니다.
            var image = Rect(name,parent,size,position).gameObject.AddComponent<UnityEngine.UI.Image>();
            image.sprite=AssetDatabase.LoadAssetAtPath<Sprite>(BasePath+"/Sprites/"+sprite+".png");
            image.raycastTarget=false;
            return image;
        }

        /// <summary>직렬화된 비공개 필드에 에디터 참조를 연결합니다.</summary>
        private static void Assign(Object target, string field, Object value)
        {
            // 런타임 필드의 접근 범위를 넓히지 않고 인스펙터 값만 설정합니다.
            var serialized=new SerializedObject(target);
            serialized.FindProperty(field).objectReferenceValue=value;
            serialized.ApplyModifiedPropertiesWithoutUndo();
        }
    }
}
#endif
